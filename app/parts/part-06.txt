  function operatePump(id){
    const p=state.pumps[id-1];
    if(p.trip){
      p.trip=false;p.mode='AUTO';state.response=Math.min(100,state.response+5);
      clearAlarm('ps'+id,true);log(`${p.label} reset and returned to AUTO after operator review.`);
      if(state.event==='pumptrip'&&state.eventPump===id){state.eventTimer=Math.min(state.eventTimer,8);}
      return render();
    }
    if(p.mode==='FORCED'){
      p.mode='AUTO';log(`${p.label} returned to AUTO.`);
    } else if(!p.run){
      p.mode='FORCED';p.run=true;state.efficiency=Math.max(0,state.efficiency-.4);
      log(`${p.label} forced to RUN to support Zone ${state.zones[p.zone].id}.`);
    }
    render();
  }

  function respondToIncident(key){
    if(key!=='event'||!state.eventMeta) return;
    const e=state.eventMeta;
    const ack=activeAlarms.get('event');
    const ackNote=ack&&ack.acked?'Alarm has been acknowledged.':'Alarm is not yet acknowledged; you can still diagnose and respond.';
    const modal=$('modal');
    const box=$('modalBox');
    box.innerHTML=`<h2>${e.name}</h2><div class="incident-context">${e.msg}<br><br><b>${ackNote}</b></div><div class="incident-options"></div><div class="row"><button class="btn" id="incidentCancel">Close</button></div>`;
    const opts=box.querySelector('.incident-options');
    e.options.forEach((op,i)=>{
      const b=document.createElement('button');
      b.className='btn incident-choice';
      b.textContent=op[0];
      b.onclick=()=>handleIncidentChoice(i);
      opts.appendChild(b);
    });
    box.querySelector('#incidentCancel').onclick=()=>modal.classList.remove('show');
    modal.classList.add('show');
  }
