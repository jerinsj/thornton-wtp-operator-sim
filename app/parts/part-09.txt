    // These do NOT represent Thornton's actual delivered chemical strengths/densities.

    const hypoEquivalentLbPerGal=6.0;
    const ammoniaEquivalentLbPerGal=4.6;

    const ferricLbDay=toLbDay(ferricDose);
    const ozoneLbDay=toLbDay(ozoneDose);
    const chlorineLbDay=toLbDay(chlorineDose);
    const ammoniaLbDay=toLbDay(ammoniaDose);

    $('ozoneDoseSP').textContent=ozoneDose.toFixed(1)+' mg/L';
    $('ozoneLbMG').textContent=toLbMG(ozoneDose).toFixed(1)+' lb/MG';
    $('ozoneLbDay').textContent=Math.round(ozoneLbDay).toLocaleString()+' lb/day';

    $('chlorineDoseSP').textContent=chlorineDose.toFixed(1)+' mg/L';
    $('chlorineLbMG').textContent=toLbMG(chlorineDose).toFixed(1)+' lb/MG';
    $('chlorineLbDay').textContent=Math.round(chlorineLbDay).toLocaleString()+' lb/day';
