  function maybeRoutineFinding(t){
    if(!['round','housekeeping','chemical','maintenance'].includes(t.type) || Math.random()>.20)return null;
    const findings=[
      'Minor pump/valve packing drip noted for maintenance follow-up.',
      'Unusual but non-critical equipment vibration/noise noted for follow-up.',
      'Small housekeeping issue corrected before it became a trip/access problem.',
      'Local indication did not exactly match the remote display; verification noted for follow-up.',
      'Chemical area inventory/housekeeping discrepancy documented for follow-up.'
    ];
    return findings[Math.floor(Math.random()*findings.length)];
  }

  function completeRoutineTask(id){
    const t=ROUTINE_TASKS.find(x=>x.id===id);
    if(!t||t.completed||state.ended)return;
    if(state.minute<t.due-45){
      showModal('Task Not Yet Due',`<p><b>${t.title}</b> is scheduled around ${shiftClock(t.due)}.</p><p>You can begin routine work when it enters its 45-minute task window.</p>`);
      return;
    }
    if(highPriorityAlarmWaiting()&&t.priority==='LOW'){
      state.response=Math.max(0,state.response-2);
      log(`Priority error: low-priority routine task '${t.title}' performed while an unacknowledged high alarm was active.`);
    }

    const before=state.minute;
    for(let i=0;i<t.duration&&!state.ended;i++)simulateMinute();
    t.completed=true;t.completedAt=state.minute;
    clearAlarm('task_'+t.id,true);
    const lateness=Math.max(0,before-t.due);
    if(lateness<=20)state.routineScore=Math.min(100,state.routineScore+1);
    t.finding=maybeRoutineFinding(t);
    const outcome=routineTaskOutcome(t);
    log(`Routine task completed: ${t.title}.`);
    if(t.finding)log(`Operator round finding: ${t.finding}`);

    showModal(t.title,`<p>${outcome}</p>${t.finding?`<p><b>Operator finding:</b> ${t.finding}</p>`:''}<p><b>Operator time used:</b> ${t.duration} simulated minutes.</p>`);
    renderRoutineTasks();render();
  }

  function updateRoutineWork(){
    ROUTINE_TASKS.forEach(t=>{
      if(t.completed)return;
      const late=state.minute-t.due;
      if(late>0){
        const hits=Math.floor(late/30)+1;
        if(hits>t.overdueHits){
          const newHits=hits-t.overdueHits;
          const weight=t.priority==='HIGH'?2.4:t.priority==='NORMAL'?1.4:.8;
          state.routineScore=Math.max(0,state.routineScore-newHits*weight);
          t.overdueHits=hits;
          if(hits===1){
            log(`Routine task overdue: ${t.title}.`);
            alarm('task_'+t.id,t.priority==='HIGH'?'med':'lo','Routine task overdue',`${t.title} was due around ${shiftClock(t.due)}. Complete it when higher-priority process/alarm work permits.`);
          }
        }
      }
    });
  }

  function bwActuatorStates(){
    const s=state.bwSeq;
    const step=s.active?s.step:-1;
    const states=[
      {name:'Influent valve',cmd:(step>=1&&step<7)?'CLOSED':'OPEN'},
      {name:'Effluent valve',cmd:(step>=1&&step<7)?'CLOSED':'OPEN'},
      {name:'Waste / drain valve',cmd:(step>=2&&step<=6)?'OPEN':'CLOSED'},
      {name:'Agitation / air valve',cmd:step===3?'OPEN':'CLOSED'},
      {name:'Backwash-water valve',cmd:(step>=4&&step<=6)?'OPEN':'CLOSED'}
    ];
    return states.map((a,idx)=>{
      let fb=a.cmd, status='ok';
      if(s.fault && s.fault.actuator===idx){
        fb=s.fault.feedback;status='fault';
      }else if(s.active && s.stepElapsed===0 && [1,2,3,4,5,6,7].includes(step)){
        status='moving';
      }
      return {...a,feedback:fb,status};
    });
  }
